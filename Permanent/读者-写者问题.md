多进程共享数据文件，读者只读不修改，写者可以读写。
- 多个读者可以**同时读**（不冲突）
- 写者和任何人**都不能同时**（写的时候别人不能读，别人读的时候不能写）

**读者优先策略**：

```
semaphore rw_mutex = 1;   // 写者互斥信号量
semaphore mutex = 1;      // 保护 read_count
int read_count = 0;       // 当前读者数量

// 写者
Writer() {
  P(rw_mutex);            // 申请写权限
    write();              // 写入数据
  V(rw_mutex);            // 释放写权限
}

// 读者
Reader() {
  P(mutex);
    read_count++;
    if (read_count == 1)    // 第一个读者
      P(rw_mutex);          // 阻止写者
  V(mutex);
  read();                   // 读取数据
  P(mutex);
    read_count--;
    if (read_count == 0)    // 最后一个读者
      V(rw_mutex);          // 允许写者
  V(mutex);
}
```

**写者优先策略**（解决读者饥饿）：

```
semaphore rw_mutex = 1;   // 写者互斥信号量
semaphore mutex = 1;      // 保护 read_count
semaphore write_mutex = 1; // 写者等待队列
int read_count = 0;

// 写者
Writer() {
  P(write_mutex);          // 写者排队
    P(rw_mutex);
      write();
    V(rw_mutex);
  V(write_mutex);
}

// 读者
Reader() {
  P(write_mutex);          // 检查是否有写者在等待
    P(mutex);
      read_count++;
      if (read_count == 1)
        P(rw_mutex);
    V(mutex);
  V(write_mutex);
  read();
  P(mutex);
    read_count--;
    if (read_count == 0)
      V(rw_mutex);
  V(mutex);
}
```
