# 手写方法的词法分析代码实现

## Rust 词法分析器实现（CRustLexer）

```cpp
// ============================================================
// RustToken 结构体（建议放在 RustLexer.h 中）
// ============================================================
struct RustToken {
    CString text;   // 词素（lexeme）文本
    CString type;   // 词素类型（关键字/标识符/宏调用/字面量/操作符/分隔符/注释）
};


// ============================================================
// RustLexer.cpp — 修复后完整实现
// ============================================================
#include "pch.h"
#include "RustLexer.h"
#include <vector>

CRustLexer::CRustLexer() {}
CRustLexer::~CRustLexer() {}

// ─── 关键字表 ───────────────────────────────────────────────
bool CRustLexer::IsKeyword(const CString& word) {
    static const CString keywords[] = {
        _T("fn"), _T("let"), _T("mut"), _T("for"), _T("in"),
        _T("if"), _T("else"), _T("match"), _T("while"), _T("return"),
        _T("struct"), _T("enum"), _T("impl"), _T("use"), _T("pub"),
        _T("as"), _T("break"), _T("const"), _T("continue"), _T("crate"),
        _T("false"), _T("loop"), _T("mod"), _T("move"), _T("ref"),
        _T("self"), _T("Self"), _T("static"), _T("trait"), _T("true"),
        _T("type"), _T("unsafe"), _T("where"), _T("dyn")
    };

    for (int i = 0; i < ARRAYSIZE(keywords); i++) {
        if (word == keywords[i]) return true;
    }
    return false;
}

// ─── 操作符表（含复合操作符 ..= 等）─────────────────────────
bool CRustLexer::IsOperator(const CString& op) {
    static const CString realOps[] = {
        _T("+"), _T("-"), _T("*"), _T("/"), _T("%"), _T("=="), _T("!="),
        _T(">"), _T("<"), _T(">="), _T("<="), _T("="), _T("+="), _T("-="),
        _T("->"), _T("=>"), _T("::"), _T("!"), _T("&"), _T("|"),
        _T(".."), _T("..=")
    };
    for (const auto& s : realOps) if (op == s) return true;
    return false;
}

// ─── 分隔符表 ───────────────────────────────────────────────
bool CRustLexer::IsPunctuation(const CString& op) {
    static const CString puncts[] = {
        _T("("), _T(")"), _T("["), _T("]"), _T("{"), _T("}"),
        _T(","), _T(";"), _T(":"), _T(".")
    };
    for (const auto& s : puncts) if (op == s) return true;
    return false;
}

// ─── 核心词法分析 ──────────────────────────────────────────
vector<RustToken> CRustLexer::Analyze(const CStringW& source) {
    vector<RustToken> tokens;
    int p = 0;                 // 当前读取位置
    int n = source.GetLength(); // 源码总长度

    while (p < n) {
        wchar_t c = source[p];

        // 1) 跳过空白字符
        if (iswspace(c)) {
            p++;
            continue;
        }

        // ──────────────────────────────────────────────
        // 2) 标识符 / 关键字 / 宏调用
        //    【修复】不再将 '!' 纳入标识符字符集
        //    → 先解析标识符，再检查后方是否紧跟 '!'
        // ──────────────────────────────────────────────
        if (iswalpha(c) || c == '_') {
            int start = p;
            while (p < n && (iswalnum(source[p]) || source[p] == '_')) {
                p++;
            }
            CString word = (CString)source.Mid(start, p - start);

            if (IsKeyword(word)) {
                tokens.push_back({ word, _T("关键字") });
            }
            else {
                // 后面紧跟 ! → 宏调用（如 vec!、println!）
                if (p < n && source[p] == '!') {
                    p++;
                    tokens.push_back({
                        (CString)source.Mid(start, p - start),
                        _T("宏调用")
                        });
                }
                else {
                    tokens.push_back({ word, _T("标识符") });
                }
            }
            continue;
        }

        // ──────────────────────────────────────────────
        // 3) 字面量（整数 / 浮点数 / 多进制）
        //    【修复】增加大写 HEX 字母 A-F、
        //    大写前缀 X/B/O、八进制前缀 o/O、
        //    以及 e/E 科学计数法检测
        // ──────────────────────────────────────────────
        if (iswdigit(c)) {
            int start = p;
            bool hasDot   = false;  // 是否包含小数点
            bool hasExp   = false;  // 是否包含 e/E 指数

            while (p < n) {
                wchar_t cur = source[p];

                if (iswdigit(cur) || cur == '_') {
                    // 数字 / 下划线 → 正常继续
                }
                else if (cur == '.') {
                    // Rust 中 .. 是范围操作符，不是浮点的一部分
                    if (p + 1 < n && source[p + 1] == '.') {
                        break;
                    }
                    hasDot = true;
                }
                else if (cur >= 'a' && cur <= 'f') {
                    // 小写 hex 字母
                }
                else if (cur >= 'A' && cur <= 'F') {
                    // 【修复 1】大写 hex 字母
                }
                else if (cur == 'e' || cur == 'E') {
                    // 【修复 5】科学计数法标记
                    hasExp = true;
                }
                else if (cur == 'x' || cur == 'X' ||
                         cur == 'b' || cur == 'B' ||
                         cur == 'o' || cur == 'O') {
                    // 【修复 2+3】进制前缀（0x, 0X, 0b, 0B, 0o, 0O）
                }
                else {
                    break; // 非法字符 → 结束数字字面量
                }
                p++;
            }

            tokens.push_back({
                (CString)source.Mid(start, p - start),
                (hasDot || hasExp) ? _T("字面量(浮点数)") : _T("字面量(整数)")
                });
            continue;
        }

        // ──────────────────────────────────────────────
        // 4) 行注释  // ...
        // ──────────────────────────────────────────────
        if (c == '/' && p + 1 < n && source[p + 1] == '/') {
            int start = p;
            while (p < n && source[p] != '\n') p++;
            tokens.push_back({
                (CString)source.Mid(start, p - start),
                _T("注释")
                });
            continue;
        }

        // ──────────────────────────────────────────────
        // 5) 字符串字面量  "..."
        //    【修复】重构为结构化 if-else，避免
        //    循环条件同时承载边界检查+转义判断
        // ──────────────────────────────────────────────
        if (c == '\"') {
            int start = p++;
            while (p < n) {
                if (source[p] == '\\') {
                    p++; // 跳过转义序列中的下一字符
                }
                else if (source[p] == '\"') {
                    p++; // 跳过结束引号
                    break;
                }
                p++;
            }
            tokens.push_back({
                (CString)source.Mid(start, p - start),
                _T("字面量(字符串)")
                });
            continue;
        }

        // ──────────────────────────────────────────────
        // 6) 操作符 / 分隔符（贪心匹配：3 字 → 2 字 → 1 字）
        // ──────────────────────────────────────────────
        auto tryMatch = [&](int len) -> bool {
            if (p + len > n) return false;
            CString op = source.Mid(p, len);
            if (IsOperator(op)) {
                tokens.push_back({ op, _T("操作符") });
                p += len;
                return true;
            }
            if (IsPunctuation(op)) {
                tokens.push_back({ op, _T("分隔符") });
                p += len;
                return true;
            }
            return false;
        };

        if (tryMatch(3)) continue;   // 优先匹配 ..=
        if (tryMatch(2)) continue;   // 然后匹配 ==, ->, ::, .. 等
        if (tryMatch(1)) continue;   // 最后匹配单个字符

        // 7) 完全无法识别的字符 → 跳过
        p++;
    }
    return tokens;
}
```

## 伪代码


```c++
while (没到结尾) {
      ch = 当前字符;

      if      (ch 是空白)    { 跳过; }
      else if (ch 是字母)    { 读完整单词 → 查关键字表 → 标类型; }
      else if (ch 是数字)    { 读完整数字 → 标类型; }
      else if (ch 是引号)    { 读到配对的引号 → 标类型; }
      else if (ch 是 '/' 且下个是 '/')  { 读到行尾 → 标类型; }
      else                   { switch(ch) 匹配运算符/分隔符; }
  
```
