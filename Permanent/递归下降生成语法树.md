# 递归下降生成语法树

> 在 [[递归下降算法]] 的基础上，将解析函数的返回类型从 `bool` 升级为 `ASTNode*`，在递归下降的同时构建抽象语法树（AST）。匹配失败时返回 `NULL`，并在回溯前清理已分配的节点内存。

## 文法

沿用之前的文法：

```
E → T | T + E
T → int | int * T | (E)
```

> 注意：该文法中 `+` 和 `*` 都是**右结合**的（如 `T → int * T` 会递归向右展开）。

// ==================== 记号定义 ====================

typedef enum {

    RPAREN,  // )
    END,     // 输入结束

typedef struct {

### AST 节点类型

```c
// ==================== AST 节点定义 ====================

typedef enum {
    NODE_INT,      // 整数字面量
    NODE_BINOP     // 二元运算（+ 或 *）
} ASTNodeType;

typedef struct ASTNode {
    ASTNodeType type;
    union {
        int intval;         // NODE_INT: 整数值
        struct {
            struct ASTNode* left;
            struct ASTNode* right;
            char op;        // NODE_BINOP: '+' 或 '*'
        } binop;
    } data;
} ASTNode;
```

## 函数实现

> 对应实际工程中的 `.c` 实现文件。

### AST 辅助函数

```c
// ==================== AST 操作函数实现 ====================

ASTNode* new_int(int val) {
    ASTNode* n = (ASTNode*)malloc(sizeof(ASTNode));
    n->type = NODE_INT;
    n->data.intval = val;
    return n;
}

ASTNode* new_binop(char op, ASTNode* left, ASTNode* right) {
    ASTNode* n = (ASTNode*)malloc(sizeof(ASTNode));
    n->type = NODE_BINOP;
    n->data.binop.op = op;
    n->data.binop.left = left;
    n->data.binop.right = right;
    return n;
}

void print_ast(const ASTNode* n, int depth) {
    if (!n) return;
    for (int i = 0; i < depth; i++) printf("  ");
    switch (n->type) {
    case NODE_INT:
        printf("Int(%d)\n", n->data.intval);
        break;
    case NODE_BINOP:
        printf("BinOp('%c')\n", n->data.binop.op);
        print_ast(n->data.binop.left,  depth + 1);
        print_ast(n->data.binop.right, depth + 1);
        break;
    }
}

void free_ast(ASTNode* n) {
    if (!n) return;
    if (n->type == NODE_BINOP) {
        free_ast(n->data.binop.left);
        free_ast(n->data.binop.right);
    }
    free(n);
}
```

### 词法分析器

```c
// ==================== 词法分析器 ====================

void lex(char* input) {
    token_count = 0;
    int i = 0;
    while (input[i] != '\0') {
        if (input[i] == ' ') { i++; continue; }
        if (input[i] == '+') {
            tokens[token_count++] = (Token){PLUS, 0};
            i++; continue;
        }
        if (input[i] == '*') {
            tokens[token_count++] = (Token){TIMES, 0};
            i++; continue;
        }
        if (input[i] == '(') {
            tokens[token_count++] = (Token){LPAREN, 0};
            i++; continue;
        }
        if (input[i] == ')') {
            tokens[token_count++] = (Token){RPAREN, 0};
            i++; continue;
        }
        if (input[i] >= '0' && input[i] <= '9') {
            int v = 0;
            while (input[i] >= '0' && input[i] <= '9') {
                v = v * 10 + (input[i] - '0');
                i++;
            }
            tokens[token_count++] = (Token){INT, v};
            continue;
        }
        tokens[token_count++] = (Token){ERR, 0};
        return;
    }
    tokens[token_count++] = (Token){END, 0};
}
```

### 解析器辅助

```c
// ==================== 解析器辅助函数 ====================

Token peek(void) {
    return tokens[next];
}

void advance(void) {
    next++;
}
```

### 非终结符 T 的产生式

```c
// ==================== T 的产生式 ====================

// T → int * T
// 优先尝试（int 是公共前缀），成功则构造 BinOp 节点
ASTNode* T_times(void) {
    int save = next;

    if (peek().type != INT) return NULL;
    int val = peek().intval;
    advance();

    if (peek().type != TIMES) { next = save; return NULL; }
    advance();

    ASTNode* right = T();
    if (!right) { next = save; return NULL; }

    return new_binop('*', new_int(val), right);
}

// T → int
ASTNode* T_int(void) {
    int save = next;

    if (peek().type != INT) return NULL;
    int val = peek().intval;
    advance();

    return new_int(val);
}

// T → (E)
ASTNode* T_paren(void) {
    int save = next;

    if (peek().type != LPAREN) return NULL;
    advance();

    ASTNode* inner = E();
    if (!inner) { next = save; return NULL; }

    if (peek().type != RPAREN) {
        free_ast(inner);
        next = save;
        return NULL;
    }
    advance();

    return inner;  // (E) 的值就是 E 的值，不产生额外节点
}

// T → int * T | int | (E)
ASTNode* T(void) {
    int save = next;
    ASTNode* n = NULL;

    if ((n = T_times())) return n;
    next = save;

    if ((n = T_int()))   return n;
    next = save;

    if ((n = T_paren())) return n;
    next = save;

    return NULL;
}
```

### 非终结符 E 的产生式

```c
// ==================== E 的产生式 ====================

// E → T
ASTNode* E_T(void) {
    return T();
}

// E → T + E
ASTNode* E_plus(void) {
    int save = next;

    ASTNode* left = T();
    if (!left) return NULL;

    if (peek().type != PLUS) {
        free_ast(left);
        next = save;
        return NULL;
    }
    advance();

    ASTNode* right = E();
    if (!right) {
        free_ast(left);
        next = save;
        return NULL;
    }

    return new_binop('+', left, right);
}

// E → T | T + E
ASTNode* E(void) {
    int save = next;
    ASTNode* n = NULL;

    if ((n = E_T()))    return n;
    next = save;

    if ((n = E_plus())) return n;
    next = save;

    return NULL;
}
```

### 入口函数

```c
// ==================== 入口 ====================

ASTNode* parse(char* input) {
    lex(input);
    if (token_count > 0 && tokens[token_count - 1].type == ERR) {
        printf("LEX ERROR: illegal character in \"%s\"\n", input);
        return NULL;
    }
    next = 0;

    ASTNode* ast = E();
    if (!ast || peek().type != END) {
        free_ast(ast);
        return NULL;
    }
    return ast;
}
```

### 测试

```c
// ==================== 测试 ====================

void test(char* input) {
    ASTNode* ast = parse(input);
    printf("%-16s => ", input);
    if (ast) {
        printf("ACCEPT\n");
        print_ast(ast, 1);
        free_ast(ast);
    } else {
        printf("REJECT\n");
    }
    printf("\n");
}

int main(void) {
    test("2");
    test("2+3");
    test("2*3");
    test("2*3+4");
    test("(2+3)*4");
    test("2+3*5");
    test("2+");      // REJECT
    test("2**3");    // REJECT
    return 0;
}
```

## 与纯识别版本的差异

| 方面 | [[递归下降算法]]（仅识别） | 本笔记（生成 AST） |
|------|---------------------------|---------------------|
| 返回类型 | `bool` | `ASTNode*` |
| 匹配失败 | `return false` | `return NULL` |
| 回溯清理 | 仅恢复 `next` | 恢复 `next` + `free_ast()` 已分配节点 |
| 匹配成功 | `return true` | 返回构造好的 `ASTNode*` |
| 结果 | 是否被语法接受 | 树形结构的语法表示 |
