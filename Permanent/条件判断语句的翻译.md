# 代码实现

if 语句的翻译需要用到[[逻辑表达式的翻译#版本 B：控制上下文（if/while 中，短路 + 回填）]]的 TC/FC 链，以及下列辅助函数（同上文定义）：

```cpp
// pair 编码："a,b"
string encodePair(int a, int b) {
    return to_string(a) + "," + to_string(b);
}
pair<int,int> decodePair(const string& s) {
    int c = s.find(',');
    return {stoi(s.substr(0, c)), stoi(s.substr(c + 1))};
}

// 链操作
int Merge(int c1, int c2);     // 合并两条链
void Backpatch(int c, int addr);  // 回填
```

### if 语句（无 else）

文法 `S → if ( BE ) S1`，**无需标记**，stmt 均推送 `"start,chain"`（起始地址 + 出口链）：

```cpp
// ═══ S → if ( BE ) S1  ═══
// 产生式编号：1（假设，实际取决于文法定义顺序）
// rhsLen = 5:  [if, (, BE, ), S1]
// attrs[4] = S1."start,chain"
// attrs[2] = BE."TC,FC"

// attrs[0..rhsLen-1] 已在主框架第 1 步中弹出

// 产生式编号约定：
//   1: S → if ( BE ) S1
//   2: S → if ( BE ) S1 N else S2
//   3: N → ε

switch (prodIdx) {
    case 1:  // S → if ( BE ) S1
    {
        auto [s1_start, s1_chain] = decodePair(attrs[4]);
        int c = attrs[2].find(',');
        int TC = stoi(attrs[2].substr(0, c));
        int FC = stoi(attrs[2].substr(c + 1));

        BackPatch(TC, stoi(s1_start));               // BE 真出口 → S1
        int chain = Merge(FC, stoi(s1_chain));        // 假出口 + S1 出口链

        result = encodePair(stoi(s1_start), chain);
        break;
    }
    case 2:  // S → if ( BE ) S1 N else S2
    {
        auto [s1_start, s1_chain] = decodePair(attrs[4]);
        auto [s2_start, s2_chain] = decodePair(attrs[7]);
        int c = attrs[2].find(',');
        int TC = stoi(attrs[2].substr(0, c));
        int FC = stoi(attrs[2].substr(c + 1));
        int jump = stoi(attrs[5]);                  // N 生成的跳转指令

        BackPatch(TC, stoi(s1_start));               // BE 真 → S1
        BackPatch(FC, stoi(s2_start));               // BE 假 → S2
        BackPatch(jump, NextStat);                   // S1 末尾 → 跳过 S2

        int chain = Merge(stoi(s1_chain), stoi(s2_chain));
        result = encodePair(stoi(s1_start), chain);
        break;
    }
    case 3:  // N → ε（标记，if-else 中 S1 与 else 之间插入）
    {
        int jump = Gen("J", "", "", 0);              // 跳过 else 分支
        result = to_string(jump);                   // N 的综合属性 = 跳转指令编号
        break;
    }
}
