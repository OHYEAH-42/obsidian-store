- 解析树从头构建
- 解析树从左到右构建
由于是最左推导，故终结符号符合中序遍历

E->T|T+E
T->int|int\*T|(E)
以这个为例
 1.  E是起始符号，从左到右推导：
	E->T->int
	 结果不符合，则回溯上一步
2.   E->T->int\*T
3.   E->T->(E)
4.   E->T->(T)
5.   E->T->(int)
最终推导出token stream要求的形式

# 伪代码实现
我们可以利用上面的，写一串伪代码来表示

前提：
一些符号我们会用TOKENs来表示，比如INT，OPEN"("，CLOSE”)“，PLUS，TIMES之类的
TOKEN是类，以上是实例
全局变量next，是一个指向当前流中token的指针

对于给定的标记TOKENs：
bool term(TOKEN tok){return next++\==tok;}
由于非终结符号一定是TOKEN，所以递归到最好next一定会自增一次

对于表达式E->T:
bool $E_{1}$ {return T()};
对于表达式E->T+E:
bool $E_{2}$ {return T()&&term(PLUS)&&E()};
对于E：
bool $E${
Token\* save=next
return(next=save,$E_{1}$||next=save,$E_{2}$)
}
save用于保存当前指针所指向的token，方便回溯
![[Pasted image 20260519232308.png]]
T也是类似的

这个算法存在一个问题，当我们的字符串为int的时候，可以使用上面的语法。
但是当我们的字符串为int\*int的时候，T()中的T1()会直接返回true，跳过了T2()，导致字符串被翻译成int。

将表达式左因子化可以解决这个问题。[[左公因子]]

# 代码实现

以下是用 C 语言实现的完整递归下降解析器，包含词法分析器和带回溯的语法解析器。

## 完整 C 语言实现

```c
#include <stdio.h>
#include <stdbool.h>
#include <string.h>
#include <stdlib.h>

// ========== 词法分析器 (Lexer) ==========

// 记号类型
typedef enum {
    INT,     // 整数字面量
    PLUS,    // +
    TIMES,   // *
    LPAREN,  // (
    RPAREN,  // )
    END,     // 输入结束
    ERR      // 非法字符
} TokenType;

typedef struct {
    TokenType type;
    int intval;  // INT 类型的值
} Token;

Token tokens[256];  // 记号流缓冲区
int token_count;    // 记号总数

// 词法分析：将输入字符串转换为记号流
void lex(char* input) {
    token_count = 0;
    int i = 0;
    while (input[i] != '\0') {
        if (input[i] == ' ') { i++; continue; }
        if (input[i] == '+') {
            tokens[token_count++] = (Token){PLUS, 0};
            i++; continue;
        }
        if (input[i] == '*') {
            tokens[token_count++] = (Token){TIMES, 0};
            i++; continue;
        }
        if (input[i] == '(') {
            tokens[token_count++] = (Token){LPAREN, 0};
            i++; continue;
        }
        if (input[i] == ')') {
            tokens[token_count++] = (Token){RPAREN, 0};
            i++; continue;
        }
        if (input[i] >= '0' && input[i] <= '9') {
            int v = 0;
            while (input[i] >= '0' && input[i] <= '9') {
                v = v * 10 + (input[i] - '0');
                i++;
            }
            tokens[token_count++] = (Token){INT, v};
            continue;
        }
        // 非法字符
        tokens[token_count++] = (Token){ERR, 0};
        return;
    }
    tokens[token_count++] = (Token){END, 0};
}

// ========== 递归下降解析器 (Parser) ==========

// 全局指针：指向当前正在处理的记号
int next;

// 匹配并消费一个记号
bool term(TokenType tt) {
    if (next < token_count && tokens[next].type == tt) {
        next++;
        return true;
    }
    return false;
}

// 前向声明
bool E();
bool T();

// ---------- T 的产生式 ----------

// T → int * T  （必须先尝试，因为 int 是公共前缀）
bool T_times() {
    int save = next;  // 保存当前位置
    if (term(INT) && term(TIMES) && T()) return true;
    next = save;      // 回溯：恢复位置
    return false;
}

// T → int
bool T_int() {
    return term(INT);
}

// T → (E)
bool T_paren() {
    int save = next;
    if (term(LPAREN) && E() && term(RPAREN)) return true;
    next = save;
    return false;
}

// T → int * T | int | (E)
// ⚠️ int*T 必须排在 int 前面，否则 int*int 会误匹配为 int 就返回
bool T() {
    int save = next;
    if (T_times()) return true;
    next = save;
    if (T_int()) return true;
    next = save;
    if (T_paren()) return true;
    next = save;
    return false;
}

// ---------- E 的产生式 ----------

// E → T + E
bool E_plus() {
    int save = next;
    if (T() && term(PLUS) && E()) return true;
    next = save;
    return false;
}

// E → T
bool E_T() {
    return T();
}

// E → T | T + E
bool E() {
    int save = next;
    if (E_T()) return true;    // 尝试 E → T
    next = save;               // 回溯
    if (E_plus()) return true; // 尝试 E → T + E
    next = save;
    return false;
}

// ========== 入口 ==========

bool parse(char* input) {
    lex(input);
    if (token_count > 0 && tokens[token_count - 1].type == ERR) {
        printf("%-20s => LEX ERROR\n", input);
        return false;
    }
    next = 0;
    bool ok = E() && term(END);
    printf("%-20s => %s\n", input, ok ? "ACCEPT" : "REJECT");
    return ok;
}

int main() {
    parse("2");
    parse("2+3");
    parse("2*3");
    parse("2*3+4");
    parse("(2+3)*4");
    parse("2+3*5");
    parse("2+");      // 不完整表达式，应 REJECT
    parse("2**3");    // 非法语法，应 REJECT
    parse("(2+3");    // 括号不匹配，应 REJECT
    return 0;
}
```

## 关键说明

### 1. 回溯的实现

每个产生式函数在执行前都会保存 `next` 指针：

```c
int save = next;
```

如果产生式匹配失败，则恢复 `next = save`，尝试下一个产生式。

### 2. 左公因子问题的处理

对于产生式 `T → int | int*T | (E)`：
- `int` 和 `int*T` 共享前缀 `int`
- **必须把 `int*T` 放在 `int` 之前尝试**，否则遇到 `int*int` 时，`T_int()` 会消费 `int` 后直接返回 `true`，剩下的 `*int` 无人处理，导致解析出错
- 这就是你在笔记中提到的"字符串为 int\*int 时，T1 直接返回 true 跳过 T2"的问题

### 3. 三种解决思路

| 方案 | 说明 |
|------|------|
| **调整尝试顺序**（本实现采用） | 将 `int*T` 放在 `int` 之前，优先尝试更长匹配 |
| **提取左公因子** | 将 `T → int \| int*T` 改写为 `T → int A`、`A → ε \| *T`，消除回溯需求 |
| **全面回溯** | 每个产生式都做 save/restore（如上实现），即便顺序不对也能通过回溯兜底，但效率较低 |

> 提取左公因子的详细方法见 [[左公因子]]
